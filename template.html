// ════════════════════════════════════════════════════════════ // NODE: processador (v3) - COM DIAGNÓSTICO ROBUSTO // Detecta e loga POR QUE linhas são descartadas // ════════════════════════════════════════════════════════════ const all = $input.all().map(i => i.json); // ══ HELPERS ══ const toISO = (s) => { if (!s) return ''; const d = new Date(s); if (isNaN(d)) return ''; return d.toISOString().slice(0, 10); }; // Parser robusto que tenta múltiplos formatos const parseDate = (s) => { if (!s) return ''; if (typeof s === 'number') { // Serial number do Excel/Sheets const utc_days = Math.floor(s - 25569); const utc_value = utc_days * 86400; const date_info = new Date(utc_value * 1000); return date_info.toISOString().slice(0, 10); } const str = String(s).trim(); if (!str) return ''; // Formato ISO: 2026-04-14 ou 2026-04-14T... if (/^\d{4}-\d{2}-\d{2}/.test(str)) { return str.slice(0, 10); } // Formato BR: dd/mm/yyyy ou d/m/yyyy if (str.includes('/')) { const parts = str.split(' ')[0].split('/'); if (parts.length === 3) { let [a, b, c] = parts; // Se tem 4 dígitos no terceiro, é dd/mm/yyyy ou m/d/yyyy if (c.length === 4) { // Heurística: se primeiro número > 12, definitivamente é dia (dd/mm/yyyy) const aNum = parseInt(a); const bNum = parseInt(b); if (aNum > 12) { // dd/mm/yyyy return `${c}-${b.padStart(2,'0')}-${a.padStart(2,'0')}`; } // Caso ambíguo (ambos <=12): assume dd/mm/yyyy (formato BR padrão da planilha) return `${c}-${b.padStart(2,'0')}-${a.padStart(2,'0')}`; } } } // Tentativa final: Date constructor const d = new Date(str); if (!isNaN(d)) return d.toISOString().slice(0, 10); return ''; }; const parseVal = (v) => { if (typeof v === 'number') return v; if (v === null || v === undefined) return 0; let s = String(v).trim(); if (!s) return 0; // formato BR: R$ 1.234,56 -> 1234.56 s = s.replace(/[R$\s]/g, ''); if (s.includes(',')) { s = s.replace(/\./g, '').replace(',', '.'); } const n = parseFloat(s); return isNaN(n) ? 0 : n; }; const cleanStr = (s) => { if (s === null || s === undefined) return ''; return String(s).trim(); }; // ══ NORMALIZAÇÃO DE ORIGENS ══ // Unifica variações como "Evento"/"Eventos", "Indicacao"/"Indicação", etc. const ORIGIN_MAP = { // Eventos 'evento': 'Eventos', 'eventos': 'Eventos', 'evento presencial': 'Eventos', 'evento online': 'Eventos', // Site 'site': 'Site', 'website': 'Site', 'site institucional': 'Site', // Redes sociais individuais (mantém separadas, mas normaliza grafia) 'instagram': 'Instagram', 'insta': 'Instagram', 'facebook': 'Facebook', 'face': 'Facebook', 'linkedin': 'LinkedIn', 'youtube': 'YouTube', 'tiktok': 'TikTok', // Mensagens 'whatsapp': 'WhatsApp', 'whats': 'WhatsApp', 'wpp': 'WhatsApp', 'email': 'E-mail', 'e-mail': 'E-mail', // Indicação 'indicacao': 'Indicação', 'indicação': 'Indicação', 'indicacoes': 'Indicação', 'indicações': 'Indicação', 'indicacões': 'Indicação', 'indicacoes ': 'Indicação', // Bases 'base leads2b': 'Base de empresas Leads2b', 'base de empresas leads2b': 'Base de empresas Leads2b', 'leads2b': 'Base de empresas Leads2b', 'base': 'Base de empresas Leads2b', // Outros canais 'google': 'Google', 'google ads': 'Google', 'meta ads': 'Meta Ads', 'rdstation': 'RD Station', 'rd station': 'RD Station', 'rd': 'RD Station', 'parceiro': 'Parceiro', 'parceiros': 'Parceiro', 'outbound': 'Outbound', 'inbound': 'Inbound', 'cold call': 'Outbound', 'prospeccao': 'Outbound', 'prospecção': 'Outbound', 'redes sociais': 'Redes Sociais', 'rede social': 'Redes Sociais' }; const normalizeOrigin = (s) => { if (!s) return 'Não informado'; const original = String(s).trim(); if (!original) return 'Não informado'; // Normaliza pra busca: minúsculas, sem acentos, sem espaços duplicados const key = original .toLowerCase() .normalize('NFD').replace(/[\u0300-\u036f]/g, '') .replace(/\s+/g, ' ') .trim(); // Lookup direto if (ORIGIN_MAP[key]) return ORIGIN_MAP[key]; // Tenta sem o "s" final (plural) if (key.endsWith('s') && ORIGIN_MAP[key.slice(0, -1)]) { return ORIGIN_MAP[key.slice(0, -1)]; } // Tenta com "s" no final (singular → plural conhecido) if (ORIGIN_MAP[key + 's']) return ORIGIN_MAP[key + 's']; // Capitalização defensiva: primeira letra maiúscula return original.charAt(0).toUpperCase() + original.slice(1); }; // ══ NORMALIZAÇÃO DE VENDEDORES ══ // Unifica variações como "Barbara B" / "Bárbara Borges" / "BARBARA" // Estratégia: usa primeiro nome como chave de fusão, mas exibe nome completo melhor disponível const USER_REGISTRY = {}; // primeiroNome (lowercase, sem acento) → nome completo canônico const normalizeKey = (s) => String(s||'').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').replace(/\s+/g, ' ').trim(); const capitalizar = (s) => { if (!s) return s; // Partículas que ficam em minúsculo no meio do nome const particulas = ['de', 'da', 'do', 'das', 'dos', 'e']; return s.split(/\s+/).map((w, i) => { if (!w) return w; const lower = w.toLowerCase(); // Primeira palavra sempre maiúscula; partículas no meio ficam minúsculas if (i > 0 && particulas.includes(lower)) return lower; return w.charAt(0).toUpperCase() + w.slice(1).toLowerCase(); }).join(' '); }; // Pre-registro de variações conhecidas (primeiroNome → nome canônico exibido) const USER_CANONICAL = { // Adicionar variações conhecidas aqui se necessário // 'barbara': 'Bárbara Borges', // 'andrey': 'Andrey Silva', }; // Construção do registry: olha todos os nomes da base e elege o "melhor" (mais completo) function buildUserRegistry(allNames) { const groups = {}; allNames.forEach(name => { const trimmed = String(name||'').trim(); if (!trimmed) return; const firstName = normalizeKey(trimmed.split(/\s+/)[0]); if (!firstName) return; if (!groups[firstName]) groups[firstName] = []; groups[firstName].push(trimmed); }); // Para cada grupo, escolhe o nome mais completo (mais palavras, mais caracteres) Object.keys(groups).forEach(firstKey => { // Se há mapeamento canônico explícito, usa if (USER_CANONICAL[firstKey]) { USER_REGISTRY[firstKey] = USER_CANONICAL[firstKey]; return; } const variants = groups[firstKey]; // Ordena: mais palavras primeiro, depois mais caracteres variants.sort((a, b) => { const wA = a.split(/\s+/).length; const wB = b.split(/\s+/).length; if (wA !== wB) return wB - wA; return b.length - a.length; }); // Escolhe o melhor: mais completo, sem abreviações tipo "Barbara B" let best = variants[0]; for (const v of variants) { const parts = v.split(/\s+/); // Se tem 2+ palavras e nenhuma é só uma letra, é melhor que abreviação const hasAbbreviation = parts.some(p => p.length === 1 || (p.length === 2 && p.endsWith('.'))); if (parts.length >= 2 && !hasAbbreviation) { best = v; break; } } // Capitaliza corretamente USER_REGISTRY[firstKey] = capitalizar(best); }); } const normalizeUser = (s) => { if (!s) return 'Sem vendedor'; const original = String(s).trim(); if (!original) return 'Sem vendedor'; const firstName = normalizeKey(original.split(/\s+/)[0]); if (!firstName) return 'Sem vendedor'; // Se está no registry, retorna o nome canônico if (USER_REGISTRY[firstName]) return USER_REGISTRY[firstName]; // Caso de borda: capitaliza e retorna return capitalizar(original); }; const LARAFY_NAMES = ['andrey', 'kaona']; const isLaraFyUser = (name) => { if (!name) return false; const n = String(name).toLowerCase(); return LARAFY_NAMES.some(x => n.includes(x)); }; const pipelineByUser = (name) => isLaraFyUser(name) ? 'LaraFy' : 'LaraTAX'; // ══ USUÁRIOS BLOQUEADOS (testes) ══ // Qualquer registro associado a esses usuários será descartado const BLOCKED_USERS = ['daniel rubio', 'gabriela samara']; const isBlockedUser = (name) => { if (!name) return false; const n = String(name).toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').trim(); return BLOCKED_USERS.some(b => n === b || n.includes(b)); }; // ══ DIAGNÓSTICO ══ const diag = { vendas: { total_brutas: 0, sem_cnpj: 0, sem_valor: 0, data_invalida: 0, aceitas: 0, exemplos_descarte: [] }, leads: { total_brutas: 0, sem_lead_id: 0, data_invalida: 0, aceitas: 0 }, opps: { total_brutas: 0, sem_opp_id: 0, data_invalida: 0, aceitas: 0 }, reunioes: { total_brutas: 0, sem_id_meet: 0, data_invalida: 0, aceitas: 0 }, atividades: { total_brutas: 0, sem_action_name: 0, data_invalida: 0, aceitas: 0 } }; // ══ 1. SEPARAR PLANILHAS ══ // LEADS: lead_name + lead_id (exclusivo) const leadsRaw = all.filter(r => r.lead_name !== undefined && r.lead_name !== null && r.lead_id !== undefined && r.lead_id !== '' && !isBlockedUser(r.lead_responsable) ); // OPS: opportunity_id + pipeline + sem lead_name const opsRaw = all.filter(r => r.opportunity_id !== undefined && r.opportunity_id !== '' && r.opportunity_id !== null && r.pipeline !== undefined && r.pipeline !== '' && (r.lead_name === undefined || r.lead_name === null) && !isBlockedUser(r.opportunity_responsible) && !isBlockedUser(r.lead_responsible) ); // VENDAS: detecta por presença de campos exclusivos da planilha de Vendas const vendasRaw = all.filter(r => { const temCNPJ = r.CNPJ !== undefined && r.CNPJ !== null && r.CNPJ !== ''; const temValor = r.valor_plano !== undefined && r.valor_plano !== null; const temCloser = r.Closer !== undefined && r.Closer !== null && r.Closer !== ''; const isVenda = (temCNPJ && temValor) || (temCloser && r.Cliente); return isVenda && !isBlockedUser(r.Closer); }); // REUNIÕES: id_meet const reunioesRaw = all.filter(r => { if (r.id_meet === undefined || r.id_meet === '') return false; // Extrai o nome do user (pode vir como JSON ou string) let userName = ''; if (r.user) { if (typeof r.user === 'string') { const match = r.user.match(/"name":"([^"]+)"/); userName = match ? match[1] : (r.user.length < 80 ? r.user : ''); } else if (r.user.name) userName = r.user.name; } return !isBlockedUser(userName); }); // ATIVIDADES: action_name + user_name_responsible (sem id_meet pra não duplicar) const atividadesRaw = all.filter(r => r.action_name !== undefined && r.user_name_responsible !== undefined && r.id_meet === undefined && !isBlockedUser(r.user_name_responsible) ); diag.vendas.total_brutas = vendasRaw.length; diag.leads.total_brutas = leadsRaw.length; diag.opps.total_brutas = opsRaw.length; diag.reunioes.total_brutas = reunioesRaw.length; diag.atividades.total_brutas = atividadesRaw.length; // ══ CONSTRUIR REGISTRY DE VENDEDORES ══ // Coleta TODOS os nomes de vendedores das 5 planilhas const todosNomes = []; leadsRaw.forEach(l => l.lead_responsable && todosNomes.push(l.lead_responsable)); opsRaw.forEach(o => { if (o.opportunity_responsible) todosNomes.push(o.opportunity_responsible); if (o.lead_responsible) todosNomes.push(o.lead_responsible); }); vendasRaw.forEach(v => v.Closer && todosNomes.push(v.Closer)); reunioesRaw.forEach(r => { if (r.user) { if (typeof r.user === 'string') { const match = r.user.match(/"name":"([^"]+)"/); if (match) todosNomes.push(match[1]); else if (r.user.length < 80) todosNomes.push(r.user); } else if (r.user.name) todosNomes.push(r.user.name); } }); atividadesRaw.forEach(a => a.user_name_responsible && todosNomes.push(a.user_name_responsible)); // Constrói o registry: agrupa por primeiro nome e escolhe o nome mais completo buildUserRegistry(todosNomes); // Diagnóstico: variantes encontradas e nome canônico escolhido const diagVendedores = {}; const todosNomesUnicos = [...new Set(todosNomes.map(n => String(n||'').trim()).filter(n => n))]; todosNomesUnicos.forEach(nome => { const canonico = normalizeUser(nome); if (!diagVendedores[canonico]) diagVendedores[canonico] = []; if (nome !== canonico && !diagVendedores[canonico].includes(nome)) { diagVendedores[canonico].push(nome); } }); // ══ 2. MAPAS DE CRUZAMENTO ══ const oppToStatus = {}; const oppToPipeline = {}; const oppToOrigin = {}; opsRaw.forEach(o => { if (o.opportunity_id) { oppToStatus[o.opportunity_id] = o.opportunity_status || 'Em andamento'; oppToPipeline[o.opportunity_id] = o.pipeline; oppToOrigin[o.opportunity_id] = normalizeOrigin(o.origin); } }); const leadToOrigin = {}; leadsRaw.forEach(l => { if (l.lead_id) leadToOrigin[l.lead_id] = normalizeOrigin(l.origin); }); // ══ 3. PROCESSAR LEADS ══ const leadsCompact = leadsRaw.map(l => { const oppId = l.opportunity_id; const oppStatus = oppId ? oppToStatus[oppId] : null; let pipe = l.pipeline; if (!pipe || pipe === '') pipe = pipelineByUser(l.lead_responsable); const d = parseDate(l.lead_created_at); if (!d) diag.leads.data_invalida++; return { d, o: normalizeOrigin(l.origin), r: normalizeUser(l.lead_responsable), uf: cleanStr(l.state) || '', p: pipe, s: cleanStr(l.lead_status) || 'Sem status', motivo: cleanStr(l.loss_reason) || '', oppId: oppId || '', oppStatus: oppStatus || '', valor: parseVal(l.order_total) }; }).filter(l => { if (l.d) { diag.leads.aceitas++; return true; } return false; }); // ══ 4. PROCESSAR OPORTUNIDADES ══ const oppsCompact = opsRaw.map(o => { let tagsArr = []; try { if (typeof o.tags === 'string' && o.tags.startsWith('[')) tagsArr = JSON.parse(o.tags); else if (Array.isArray(o.tags)) tagsArr = o.tags; else if (typeof o.tags === 'string' && o.tags) tagsArr = o.tags.split(',').map(t => t.trim()).filter(t => t); } catch(e) { tagsArr = []; } const d = parseDate(o.opportunity_created_at); if (!d) diag.opps.data_invalida++; return { d, s: cleanStr(o.opportunity_status) || 'Em andamento', step: cleanStr(o.pipeline_step) || 'Sem etapa', v: parseVal(o.total_value), o: normalizeOrigin(o.origin), r: normalizeUser(o.opportunity_responsible || o.lead_responsible), c: cleanStr(o.contact_name).slice(0, 60), p: cleanStr(o.pipeline) || 'Sem pipeline', dias: parseInt(o.days_in_pipeline) || 0, motivo: cleanStr(o.loss_reason) || '', detalhes: cleanStr(o.loss_details).slice(0, 100), tags: tagsArr, leadId: o.lead_id || '' }; }).filter(o => { if (o.d) { diag.opps.aceitas++; return true; } return false; }); // ══ 5. PROCESSAR VENDAS - COM DIAGNÓSTICO DETALHADO ══ const vendasCompact = vendasRaw.map((v, idx) => { const valMensal = parseVal(v.valor_plano); const valAnual = parseVal(v['Valor anual']) || (valMensal * 12); const impl = cleanStr(v.implantacao); const dataAss = v['Data assinatura'] || v['período'] || v['period']; const d = parseDate(dataAss); // Diagnóstico if (!d) { diag.vendas.data_invalida++; if (diag.vendas.exemplos_descarte.length < 5) { diag.vendas.exemplos_descarte.push({ motivo: 'Data inválida', cliente: cleanStr(v.Cliente).slice(0, 30), data_recebida: dataAss, tipo_data: typeof dataAss }); } } if (!v.CNPJ) diag.vendas.sem_cnpj++; if (valMensal === 0 && valAnual === 0) diag.vendas.sem_valor++; return { d, cl: normalizeUser(v.Closer), v: valMensal, va: valAnual, o: normalizeOrigin(v.Lead), tl: cleanStr(v.tipo_lead) || '', i: impl === 'Sim' ? 'Sim' : (impl === 'Não' || impl === 'Nao' ? 'Não' : 'Não'), s: cleanStr(v['status contrato']) || 'Ativo', c: cleanStr(v.Cliente).slice(0, 60), cnpj: cleanStr(v.CNPJ), uf: cleanStr(v.UF_cliente) || '', pl: cleanStr(v['Descrição']) || '', df: parseDate(v['Data de término do contrato']) }; }).filter(v => { if (v.d) { diag.vendas.aceitas++; return true; } return false; }); // ══ 6. PROCESSAR REUNIÕES ══ const reunioesCompact = reunioesRaw.map(r => { const durSec = parseInt(r.duration) || 0; const durMin = Math.round(durSec / 60); const statusRaw = cleanStr(r.status).toLowerCase(); const statusFinal = statusRaw === 'done' ? 'done' : 'pending'; let userName = ''; if (r.user) { if (typeof r.user === 'string') { const match = r.user.match(/"name":"([^"]+)"/); userName = match ? match[1] : (r.user.length > 80 ? 'Sem vendedor' : r.user); } else if (r.user.name) userName = r.user.name; } if (!userName) userName = 'Sem vendedor'; const pipe = pipelineByUser(userName); let origem = oppToOrigin[r.id_entity] || leadToOrigin[r.id_entity] || 'Não informado'; const d = parseDate(r.start); if (!d) diag.reunioes.data_invalida++; return { d, u: normalizeUser(userName), dur: durMin, s: statusFinal, sub: cleanStr(r.subject).slice(0, 60), p: pipe, o: origem, cn: cleanStr(r.company_name).slice(0, 60), e: cleanStr(r.entity) || '', idEnt: cleanStr(r.id_entity) || '' }; }).filter(r => { if (r.d) { diag.reunioes.aceitas++; return true; } return false; }); // ══ 7. PROCESSAR ATIVIDADES ══ const atividadesCompact = atividadesRaw.map(a => { const userName = cleanStr(a.user_name_responsible) || 'Sem vendedor'; const pipe = pipelineByUser(userName); const fechada = !!(a.close_date && cleanStr(a.close_date) !== ''); let origem = oppToOrigin[a.id_entity] || leadToOrigin[a.id_entity] || 'Não informado'; const d = parseDate(a.action_date || a.created_at); if (!d) diag.atividades.data_invalida++; return { d, u: normalizeUser(userName), n: cleanStr(a.action_name).slice(0, 60), t: cleanStr(a.type) || '', e: fechada, p: pipe, o: origem, msg: cleanStr(a.message).slice(0, 80), ent: cleanStr(a.entity) || '', idEnt: cleanStr(a.id_entity) || '' }; }).filter(a => { if (a.d) { diag.atividades.aceitas++; return true; } return false; }); // ══ 8. LISTAS ÚNICAS ══ const allVend = new Set(); leadsCompact.forEach(l => l.r && l.r !== 'Sem responsável' && allVend.add(l.r)); oppsCompact.forEach(o => o.r && o.r !== 'Sem responsável' && allVend.add(o.r)); vendasCompact.forEach(v => v.cl && v.cl !== 'Sem vendedor' && allVend.add(v.cl)); reunioesCompact.forEach(r => r.u && r.u !== 'Sem vendedor' && allVend.add(r.u)); atividadesCompact.forEach(a => a.u && a.u !== 'Sem vendedor' && allVend.add(a.u)); const allOrigin = new Set(); leadsCompact.forEach(l => l.o && l.o !== 'Não informado' && allOrigin.add(l.o)); oppsCompact.forEach(o => o.o && o.o !== 'Não informado' && allOrigin.add(o.o)); vendasCompact.forEach(v => v.o && v.o !== 'Não informado' && allOrigin.add(v.o)); const allVendaStatus = new Set(); vendasCompact.forEach(v => v.s && allVendaStatus.add(v.s)); const allPipelineSteps = new Set(); oppsCompact.forEach(o => o.step && o.step !== 'Sem etapa' && allPipelineSteps.add(o.step)); const allLeadStatus = new Set(); leadsCompact.forEach(l => l.s && l.s !== 'Sem status' && allLeadStatus.add(l.s)); // ══ 9. METADADOS ══ const now = new Date(); const todayISO = now.toISOString().slice(0, 10); const allDates = [ ...leadsCompact.map(l => l.d), ...oppsCompact.map(o => o.d), ...vendasCompact.map(v => v.d), ...reunioesCompact.map(r => r.d), ...atividadesCompact.map(a => a.d) ].filter(d => d).sort(); const dataAtualizacao = now.toLocaleDateString('pt-BR', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric', hour: '2-digit', minute: '2-digit', timeZone: 'America/Sao_Paulo' }); // ══ 10. RETORNO ══ return [{ json: { dataAtualizacao, minDate: allDates[0] || todayISO, maxDate: todayISO, counts: { leads: leadsCompact.length, opps: oppsCompact.length, vendas: vendasCompact.length, reunioes: reunioesCompact.length, atividades: atividadesCompact.length }, diagnostico: diag, diagVendedores, leads: leadsCompact, opps: oppsCompact, vendas: vendasCompact, reunioes: reunioesCompact, atividades: atividadesCompact, filtros: { vendedores: [...allVend].sort(), origens: [...allOrigin].sort(), vendaStatus: [...allVendaStatus].sort(), pipelineSteps: [...allPipelineSteps].sort(), leadStatus: [...allLeadStatus].sort() } } }];
